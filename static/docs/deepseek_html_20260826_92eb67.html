Request & Response

Generate

POST /generate
Content-Type: application/json

{
  "prompt": "A beautiful sunset over mountains",
  "ratio": "16:9",
  "format": "jpg"
}

Returns the generated image as binary data.

Edit

POST /edit
Content-Type: application/json

{
  "image": "base64_string_without_prefix",
  "prompt": "Make this image more realistic",
  "format": "png"
}

Returns the edited image as binary data.

Ignore Reference

POST /ignore_ref
Content-Type: application/json

{
  "prompt": "A futuristic city at night",
  "ratio": "16:9",
  "format": "jpg"
}

Returns the generated image as binary data.

Code Examples

JavaScript

class ImageGenAPI {
  constructor(baseUrl = 'https://raihan07-geminiapi.onrender.com') {
    this.baseUrl = baseUrl;
  }

  async generate(prompt, ratio = '9:16', format = 'jpg') {
    const res = await fetch(`${this.baseUrl}/generate`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ prompt, ratio, format })
    });
    if (!res.ok) throw new Error((await res.json()).error);
    return await res.blob();
  }

  async edit(imageBase64, prompt, format = 'jpg') {
    const res = await fetch(`${this.baseUrl}/edit`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ image: imageBase64, prompt, format })
    });
    if (!res.ok) throw new Error((await res.json()).error);
    return await res.blob();
  }
}

Python

import requests
import base64

class ImageGenAPI:
    def __init__(self, base_url='https://raihan07-geminiapi.onrender.com'):
        self.base_url = base_url

    def generate(self, prompt, ratio='9:16', format='jpg'):
        resp = requests.post(f'{self.base_url}/generate',
            json={'prompt': prompt, 'ratio': ratio, 'format': format})
        resp.raise_for_status()
        return resp.content

    def edit(self, image_path, prompt, format='jpg'):
        with open(image_path, 'rb') as f:
            b64 = base64.b64encode(f.read()).decode()
        resp = requests.post(f'{self.base_url}/edit',
            json={'image': b64, 'prompt': prompt, 'format': format})
        resp.raise_for_status()
        return resp.content
